    \node[font=\tiny, text=evinktwo, anchor=east, inner sep=1pt] at (-0.56,0.21) {real};
    \node[font=\tiny, text=evinktwo, anchor=east, inner sep=1pt] at (-0.56,-0.21) {fake};
    % real row: TN, FP
    \ifnum#3=2
      \draw[evinktwo!50, dashed, fill=white, rounded corners=1.5pt] (-0.54,0.01) rectangle (0.54,0.43);
      \node[font=\tiny, text=evinktwo] at (0,0.22) {none};
    \else
      \fill[#2!18!white, rounded corners=1.5pt] (-0.54,0.01) rectangle (-0.01,0.43);
      \fill[#2, rounded corners=1.5pt] (0.01,0.01) rectangle (0.54,0.43);
      \node[font=\evsmall] at (-0.27,0.22) {TN};
      \node[font=\evsmall\bfseries, text=white] at (0.27,0.22) {FP};
    \fi
    % fake row: FN, TP
    \ifnum#3=1

      \fill[#2!18!white, rounded corners=1.5pt] (0.01,-0.43) rectangle (0.54,-0.01);
      \node[font=\evsmall\bfseries, text=white] at (-0.27,-0.22) {FN};
      \node[font=\evsmall] at (0.27,-0.22) {TP};
    \fi
  \end{scope}}
