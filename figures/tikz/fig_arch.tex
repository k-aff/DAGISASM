%  Figure 4: architecture of the multimodal detector (TikZ)

%  Portrait page. The two streams run left to right, then both feature
%  vectors drop down to the concatenation and the flow continues
%  downwards: fusion, embedding z, the two heads, then detail boxes.

%  Use inside a normal figure environment:
%      \begin{figure}[p]
%      \centering
%      \input{figures/fig_arch.tex}
%      \caption{...}\label{fig:arch}

%  Needs in the preamble:
%      \usepackage{tikz}
%      \usetikzlibrary{positioning,fit,backgrounds,arrows.meta,calc}
%
%  Drawing conventions
%    * a block is a tensor: front face = spatial size, depth = channels
%      (log scale); grid lines on the Swin blocks are attention windows
%      (8 x 8 windows at 56^2, 4 x 4 at 28^2, 2 x 2 at 14^2, one at 7^2)
%    * a row or column of cells is a vector, one cell per 64 values
%    * shapes are channels x spatial size, so 192 x 28^2 = 192 x 28 x 28
%  Values follow the model code (detector.py, freq.py, grl.py, trainer.py)
%  and timm's swin_tiny_patch4_window7_224 (depths 2-2-6-2, heads 3-6-12-24).
%  Colours follow the poster: blue = image stream, purple = frequency
%  stream, magenta = fusion and forgery head, teal = race adversary.
%  Coordinates are scaled (x 1.1, y 1.02) to give the parts more room,
%  so the canvas is about 15.1 cm x 15.9 cm. The picture is centred in a
%  box of text width and reaches about 0.7 cm into each margin.
% =====================================================================

% \archcube{(centre)}{front size}{depth}{colour}{windows per side}
\providecommand{\archcube}[5]{%
  \begin{scope}[shift={#1}]
    \pgfmathsetmacro{\arS}{#2}\pgfmathsetmacro{\arH}{#2/2}%
    \pgfmathsetmacro{\arDX}{0.55*#3}\pgfmathsetmacro{\arDY}{0.40*#3}%
    \begin{scope}[shift={(-\arDX/2,-\arDY/2)}, line width=0.3pt, line join=round]
      \filldraw[fill=#4!88!black, draw=#4!60!black]
        (\arH,-\arH) -- ++(\arDX,\arDY) -- ++(0,\arS) -- (\arH,\arH) -- cycle;
      \filldraw[fill=#4!62!white, draw=#4!60!black]
        (-\arH,\arH) -- ++(\arDX,\arDY) -- ++(\arS,0) -- (\arH,\arH) -- cycle;
      \filldraw[fill=#4!30!white, draw=#4!60!black] (-\arH,-\arH) rectangle (\arH,\arH);
      \ifnum#5>1
        \pgfmathsetmacro{\arStep}{#2/#5}\pgfmathtruncatemacro{\arN}{#5-1}%
        \foreach \k in {1,...,\arN}{%
          \draw[white, line width=0.35pt] (-\arH+\k*\arStep,-\arH) -- ++(0,\arS);
          \draw[white, line width=0.35pt] (-\arH,-\arH+\k*\arStep) -- ++(\arS,0);
        }%
      \fi
    \end{scope}
  \end{scope}}
% \archvec{(centre)}{cells}{colour}: horizontal vector, cells of 0.12 cm

  \end{scope}}
